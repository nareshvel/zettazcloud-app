# RBAC API Integration Guide

## Authentication Flow

The RBAC system will enhance the authentication flow by including role and permission data in the JWT token.

## Backend Integration

### Permission Middleware

```javascript
const checkPermission = (requiredPermission, scopeType = null) => {
  return async (req, res, next) => {
    // Skip check if no permission required
    if (!requiredPermission) return next();

    try {
      const { userId, tenantId } = req.user;
      const storeId = req.params.storeId || req.query.storeId || req.body.storeId;

      // Fast check if permission is in token
      if (req.user.permissions && req.user.permissions.includes(requiredPermission)) {
        return next();
      }

      // Otherwise query database for fresh permissions
      const userPerms = await getUserPermissions(userId, tenantId, storeId);
      
      if (userPerms.includes(requiredPermission)) {
        return next();
      }
      
      return res.status(403).json({
        success: false,
        message: 'Permission denied',
        error: `You don't have permission to perform this action: ${requiredPermission}`
      });
    } catch (error) {
      console.error('Permission check error:', error);
      return res.status(500).json({
        success: false,
        message: 'Internal server error during permission check'
      });
    }
  };
};
```

### Permission Helper Functions

```javascript
/**
 * Get permissions for a user in a specific context (tenant or store)
 */
const getUserPermissions = async (userId, tenantId, storeId = null) => {
  // Query to fetch all permissions for this user
  const query = `
    SELECT DISTINCT p.name
    FROM permissions p
    JOIN role_permissions rp ON p.id = rp.permission_id
    JOIN user_roles ur ON rp.role_id = ur.role_id
    WHERE ur.user_id = ?
    AND (
      (ur.scope = 'tenant' AND ur.scope_id = ?)
      OR (ur.scope = 'store' AND ur.scope_id = ?)
    )
  `;

  const [rows] = await db.query(query, [userId, tenantId, storeId]);
  return rows.map(row => row.name);
};

/**
 * Check if user has specific permission
 */
const hasPermission = async (userId, permissionName, tenantId, storeId = null) => {
  const permissions = await getUserPermissions(userId, tenantId, storeId);
  return permissions.includes(permissionName);
};
```

## Frontend Integration

### Updated Auth Context

```tsx
// src/context/AuthContext.tsx
import React, { createContext, useState, useEffect, useContext } from 'react';
import { fetchApi } from '../utils/api';

type Scope = 'tenant' | 'store';

interface UserRole {
  roleId: string;
  name: string;
  scope: Scope;
  scopeId: string;
}

interface AuthContextType {
  isAuthenticated: boolean;
  user: any;
  roles: UserRole[];
  permissions: string[];
  currentStoreId: string | null;
  login: (email: string, password: string) => Promise<void>;
  logout: () => void;
  setCurrentStore: (storeId: string) => void;
  hasPermission: (permission: string) => boolean;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [isAuthenticated, setIsAuthenticated] = useState<boolean>(false);
  const [user, setUser] = useState<any>(null);
  const [roles, setRoles] = useState<UserRole[]>([]);
  const [permissions, setPermissions] = useState<string[]>([]);
  const [currentStoreId, setCurrentStoreId] = useState<string | null>(null);

  // Check if token exists on load
  useEffect(() => {
    const token = localStorage.getItem('token');
    if (token) {
      validateToken();
    }
  }, []);

  const validateToken = async () => {
    try {
      const userData = await fetchApi('/api/auth/me');
      setIsAuthenticated(true);
      setUser(userData);
      setRoles(userData.roles || []);
      setPermissions(userData.permissions || []);
      
      // Set default store if available
      if (userData.defaultStoreId) {
        setCurrentStoreId(userData.defaultStoreId);
      }
    } catch (error) {
      console.error('Token validation failed:', error);
      logout();
    }
  };

  const login = async (email: string, password: string) => {
    try {
      const result = await fetchApi('/api/auth/login', {
        method: 'POST',
        body: JSON.stringify({ email, password }),
      });
      
      localStorage.setItem('token', result.token);
      setIsAuthenticated(true);
      setUser(result.user);
      setRoles(result.user.roles || []);
      setPermissions(result.user.permissions || []);
      
      // Set default store if available
      if (result.user.defaultStoreId) {
        setCurrentStoreId(result.user.defaultStoreId);
      }
    } catch (error) {
      console.error('Login failed:', error);
      throw error;
    }
  };

  const logout = () => {
    localStorage.removeItem('token');
    setIsAuthenticated(false);
    setUser(null);
    setRoles([]);
    setPermissions([]);
    setCurrentStoreId(null);
  };

  const setCurrentStore = (storeId: string) => {
    setCurrentStoreId(storeId);
    // Could optionally refresh permissions based on the store context
  };

  const hasPermission = (permission: string): boolean => {
    return permissions.includes(permission);
  };

  return (
    <AuthContext.Provider value={{
      isAuthenticated,
      user,
      roles,
      permissions,
      currentStoreId,
      login,
      logout,
      setCurrentStore,
      hasPermission
    }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = (): AuthContextType => {
  const context = useContext(AuthContext);
  if (context === undefined) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
};
```

### Permission Component

```tsx
// src/components/Permission.tsx
import React from 'react';
import { useAuth } from '../context/AuthContext';

interface PermissionProps {
  require: string | string[];
  fallback?: React.ReactNode;
  children: React.ReactNode;
}

export const Permission: React.FC<PermissionProps> = ({
  require,
  fallback = null,
  children
}) => {
  const { hasPermission } = useAuth();

  // Handle array of permissions (ANY logic)
  if (Array.isArray(require)) {
    const hasAnyPermission = require.some(permission => hasPermission(permission));
    return hasAnyPermission ? <>{children}</> : <>{fallback}</>;
  }

  // Handle single permission
  return hasPermission(require) ? <>{children}</> : <>{fallback}</>;
};

// Usage example:
// <Permission require="products.create">
//   <button>Create Product</button>
// </Permission>
```

### Store Selector Component

```tsx
// src/components/StoreSelector.tsx
import React, { useState, useEffect } from 'react';
import { useAuth } from '../context/AuthContext';
import { fetchApi } from '../utils/api';

interface Store {
  id: string;
  name: string;
  roleNames: string[];
}

export const StoreSelector: React.FC = () => {
  const { currentStoreId, setCurrentStore } = useAuth();
  const [stores, setStores] = useState<Store[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchStores = async () => {
      try {
        setLoading(true);
        const response = await fetchApi('/api/user/stores');
        setStores(response.stores || []);
        setLoading(false);
      } catch (error) {
        console.error('Failed to fetch stores:', error);
        setLoading(false);
      }
    };

    fetchStores();
  }, []);

  if (loading) {
    return <div>Loading stores...</div>;
  }

  if (stores.length === 0) {
    return <div>No stores available</div>;
  }

  return (
    <div className="store-selector">
      <label htmlFor="store-select">Select Store:</label>
      <select
        id="store-select"
        value={currentStoreId || ''}
        onChange={(e) => setCurrentStore(e.target.value)}
      >
        <option value="" disabled>
          Select a store
        </option>
        {stores.map((store) => (
          <option key={store.id} value={store.id}>
            {store.name} ({store.roleNames.join(', ')})
          </option>
        ))}
      </select>
    </div>
  );
};
```

## API Endpoints

### Role Management

```
GET    /api/roles                 - List roles
POST   /api/roles                 - Create role
GET    /api/roles/:id             - Get role details
PUT    /api/roles/:id             - Update role
DELETE /api/roles/:id             - Delete role
GET    /api/roles/:id/permissions - List role permissions
PUT    /api/roles/:id/permissions - Update role permissions
```

### User Role Management

```
GET    /api/users/:id/roles                 - List user roles
POST   /api/users/:id/roles                 - Assign role to user
DELETE /api/users/:id/roles/:roleId         - Remove role from user
GET    /api/users/:id/permissions           - List user permissions
GET    /api/users/:id/stores                - List user's accessible stores
```

### Permission Management

```
GET    /api/permissions           - List all permissions
GET    /api/permissions/modules   - List permission modules
```

## Testing and Validation

### Permission Checking Logic

```javascript
// Unit test for permission middleware
describe('Permission Middleware', () => {
  it('should allow access when user has required permission', async () => {
    // Setup
    const req = {
      user: {
        userId: 'test-user-id',
        tenantId: 'test-tenant-id',
        permissions: ['products.view']
      }
    };
    const res = {};
    const next = jest.fn();
    
    // Execute
    await checkPermission('products.view')(req, res, next);
    
    // Assert
    expect(next).toHaveBeenCalled();
  });
  
  it('should deny access when user lacks required permission', async () => {
    // Setup
    const req = {
      user: {
        userId: 'test-user-id',
        tenantId: 'test-tenant-id',
        permissions: ['products.view']
      }
    };
    const res = {
      status: jest.fn().mockReturnThis(),
      json: jest.fn()
    };
    const next = jest.fn();
    
    // Execute
    await checkPermission('products.edit')(req, res, next);
    
    // Assert
    expect(next).not.toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(403);
    expect(res.json).toHaveBeenCalledWith(
      expect.objectContaining({
        success: false,
        message: 'Permission denied'
      })
    );
  });
});
```
