# Removing the Legacy Role Column: Implementation Guide

This document provides detailed implementation examples for removing the legacy `role` column from the `users` table and fully transitioning to the RBAC system. It complements the migration plan with specific code changes required.

## Database Migration Script

Create a new migration file in the `backend/migrations` directory:

```javascript
// backend/migrations/20250625_remove_role_column.js

/**
 * @param {import('mysql2/promise').Connection} db
 */
exports.up = async function(db) {
  console.log('Starting migration: Remove legacy role column');
  
  try {
    // First ensure all users have proper RBAC roles based on their legacy role
    await db.query(`
      INSERT IGNORE INTO user_roles (id, user_id, role_id, scope, store_id, created_at, updated_at)
      SELECT 
          UUID() as id,
          u.id as user_id,
          (SELECT id FROM roles WHERE name = u.role AND tenant_id = u.tenant_id LIMIT 1) as role_id,
          CASE WHEN u.store_id IS NULL THEN 'tenant' ELSE 'store' END as scope,
          u.store_id,
          NOW() as created_at,
          NOW() as updated_at
      FROM users u
      WHERE NOT EXISTS (
          SELECT 1 FROM user_roles ur WHERE ur.user_id = u.id
      );
    `);
    
    // Verify all users have at least one role
    const [usersWithNoRoles] = await db.query(`
      SELECT u.id, u.name, u.email, u.role 
      FROM users u
      LEFT JOIN user_roles ur ON u.id = ur.user_id
      WHERE ur.id IS NULL;
    `);
    
    if (usersWithNoRoles.length > 0) {
      console.warn(`Warning: Found ${usersWithNoRoles.length} users without RBAC roles. Migration halted.`);
      console.warn('Please assign roles to these users before proceeding.');
      throw new Error('Users without RBAC roles detected');
    }
    
    // If all users have roles, proceed with removing the column
    await db.query(`ALTER TABLE users DROP COLUMN role;`);
    
    console.log('Migration successful: Removed legacy role column');
    
  } catch (error) {
    console.error('Migration failed:', error);
    throw error;
  }
};

/**
 * @param {import('mysql2/promise').Connection} db
 */
exports.down = async function(db) {

  } catch (error) {
    console.error('Rollback failed:', error);
    throw error;
  }
```

```javascript
/**
```

## Update User Role Service

Add a new method to fetch all roles and permissions for a user:

```javascript
// backend/services/userRoleService.js

/**
 * Get all roles and permissions for a user
 * @param {String} userId - User ID
 * @returns {Promise<Object>} Object with roles and permissions arrays
 */
const getUserRolesAndPermissions = async (userId) => {
  const connection = await db.getConnection();
  
  try {
    // Get all roles for the user
    const [userRoles] = await connection.query(`
      SELECT r.id, r.name, r.description, ur.scope, ur.store_id
      FROM user_roles ur
      JOIN roles r ON ur.role_id = r.id
      WHERE ur.user_id = ?
    `, [userId]);
    
    // Get all permissions for these roles
    const roleIds = userRoles.map(r => r.id);
    
    if (!roleIds.length) {
      return { roles: [], permissions: [] };
    }
    
    const [permissions] = await connection.query(`
      SELECT DISTINCT p.name
      FROM role_permissions rp
      JOIN permissions p ON rp.permission_id = p.id
      WHERE rp.role_id IN (?)
    `, [roleIds]);
    
    return {
      roles: userRoles,
      permissions: permissions.map(p => p.name)
    };
    
  } finally {
    connection.release();
  }
};

module.exports = {

  getUserRolesAndPermissions
};
```

## Update Authorization Middleware

Replace the role-based checks with permission-based checks:

```javascript
// backend/middleware/authMiddleware.js

/**
 * Check if the user has the required permission
 * @param {Array} requiredPermissions - Array of required permissions
 * @returns {Function} Express middleware
 */
const hasPermission = (requiredPermissions) => {
  return (req, res, next) => {
    // Skip checks in development if flag is set
    if (process.env.NODE_ENV === 'development' && process.env.SKIP_AUTH_CHECKS === 'true') {
      return next();
    }
    
    if (!req.user) {
      return res.status(401).json({ message: 'Authentication required' });
    }
    
    const userPermissions = req.user.permissions || [];
    
    // Check if user has any of the required permissions
    const hasRequired = requiredPermissions.some(perm => userPermissions.includes(perm));
    
    if (hasRequired) {
      return next();
    }
    
    // If no required permissions matched
    return res.status(403).json({ message: 'Permission denied' });
  };
};
```

## Update User Creation Route

Remove the legacy role field and use only RBAC roles:

```javascript
// backend/routes/userRoutes.js

/**
 * @route   POST /api/users
 * @desc    Create a new user
 * @access  Private (requires USER_CREATE permission)
 */
router.post('/', [authenticate, hasPermission([PERMISSIONS.USER_CREATE])], async (req, res) => {
  const { name, email, password, roles, phone_number, store_id } = req.body;
  const tenant_id = req.user.tenant_id;
  
  const connection = await pool.getConnection();
  try {
    await connection.beginTransaction();
    
    // Basic validation
    if (!name || !email || !password || !roles || !Array.isArray(roles) || !roles.length) {
      return res.status(400).json({ 
        message: 'Please provide name, email, password, and at least one role' 
      });
    }
    
    // Check if user already exists
    const [existingUsers] = await connection.query(
      'SELECT id FROM users WHERE email = ? AND tenant_id = ?',
      [email, tenant_id]
    );
    
    if (existingUsers.length > 0) {
      return res.status(409).json({ message: 'User with this email already exists' });
    }
    
    // Hash password
    const salt = await bcrypt.genSalt(10);
    const hashedPassword = await bcrypt.hash(password, salt);
    
    // Normalize store_id (empty string → null)
    const primaryStoreId = store_id === '' ? null : store_id;
    
    // Create user with basic info
    const userId = uuid();
    const newUser = {
      id: userId,
      tenant_id,
      name,
      email,
      password_hash: hashedPassword,
      phone_number: phone_number || null,
      store_id: primaryStoreId,
      created_at: new Date(),
      updated_at: new Date()
    };
    
    await connection.query('INSERT INTO users SET ?', [newUser]);
    
    // Assign roles
    for (const role of roles) {
      try {
        await userRoleService.assignTenantRole(
          userId,
          role.id,
          tenant_id,
          role.scope || 'tenant',
          role.store_id === '' ? null : role.store_id,
          req.user.id // assigned_by
        );
      } catch (roleError) {
        console.error(`Error assigning role ${role.id} to user ${userId}:`, roleError);
        // Continue with other roles
      }
    }
    
    await connection.commit();
    res.status(201).json({ message: 'User created successfully', userId });
    
  } catch (error) {
    await connection.rollback();
    console.error('Error creating user:', error);
    res.status(500).json({ message: 'Server error while creating user' });
  } finally {
    connection.release();
  }
});
```
