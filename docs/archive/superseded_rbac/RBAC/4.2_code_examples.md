# RBAC Code Examples

This document provides practical code examples for implementing the Role-Based Access Control system in both backend and frontend components.

## Backend Implementation Examples

### Permission Service

```javascript
// services/permissionService.js

const db = require('../config/database');

class PermissionService {
  /**
   * Get all available permissions
   */
  async getAllPermissions() {
    const [rows] = await db.query('SELECT * FROM permissions ORDER BY module, name');
    return rows;
  }

  /**
   * Group permissions by module
   */
  async getPermissionsByModule() {
    const permissions = await this.getAllPermissions();
    
    return permissions.reduce((modules, permission) => {
      if (!modules[permission.module]) {
        modules[permission.module] = [];
      }
      modules[permission.module].push(permission);
      return modules;
    }, {});
  }

  /**
   * Get permissions for a specific role
   */
  async getRolePermissions(roleId) {
    const query = `
      SELECT p.*
      FROM permissions p
      JOIN role_permissions rp ON p.id = rp.permission_id
      WHERE rp.role_id = ?
      ORDER BY p.module, p.name
    `;
    
    const [rows] = await db.query(query, [roleId]);
    return rows;
  }

  /**
   * Get user permissions for a specific context (tenant or store)
   */
  async getUserPermissions(userId, tenantId, storeId = null) {
    let query = `
      SELECT DISTINCT p.name
      FROM permissions p
      JOIN role_permissions rp ON p.id = rp.permission_id
      JOIN user_roles ur ON rp.role_id = ur.role_id
      JOIN roles r ON ur.role_id = r.id
      WHERE ur.user_id = ?
      AND r.tenant_id = ?
    `;
    
    // If store context is provided, include store-specific roles
    if (storeId) {
      query += `
        AND (
          ur.scope = 'tenant'
          OR (ur.scope = 'store' AND ur.scope_id = ?)
        )
      `;
      const [rows] = await db.query(query, [userId, tenantId, storeId]);
      return rows.map(row => row.name);
    } else {
      // Only tenant-wide roles
      query += ` AND ur.scope = 'tenant'`;
      const [rows] = await db.query(query, [userId, tenantId]);
      return rows.map(row => row.name);
    }
  }

  /**
   * Check if a user has a specific permission
   */
  async hasPermission(userId, permissionName, tenantId, storeId = null) {
    const permissions = await this.getUserPermissions(userId, tenantId, storeId);
    return permissions.includes(permissionName);
  }
}

module.exports = new PermissionService();
```

### Role Service

```javascript
// services/roleService.js

const { v4: uuidv4 } = require('uuid');
const db = require('../config/database');

class RoleService {
  /**
   * Get all roles for a tenant
   */
  async getRoles(tenantId) {
    const [rows] = await db.query(
      'SELECT * FROM roles WHERE tenant_id = ? ORDER BY name',
      [tenantId]
    );
    return rows;
  }

  /**
   * Get a specific role
   */
  async getRole(roleId, tenantId) {
    const [rows] = await db.query(
      'SELECT * FROM roles WHERE id = ? AND tenant_id = ?',
      [roleId, tenantId]
    );
    return rows[0] || null;
  }

  /**
   * Create a new role
   */
  async createRole(tenantId, roleData) {
    const roleId = uuidv4();
    const { name, description } = roleData;
    
    await db.query(
      'INSERT INTO roles (id, tenant_id, name, description) VALUES (?, ?, ?, ?)',
      [roleId, tenantId, name, description]
    );
    
    return this.getRole(roleId, tenantId);
  }

  /**
   * Update a role
   */
  async updateRole(roleId, tenantId, roleData) {
    const { name, description } = roleData;
    
    await db.query(
      'UPDATE roles SET name = ?, description = ? WHERE id = ? AND tenant_id = ?',
      [name, description, roleId, tenantId]
    );
    
    return this.getRole(roleId, tenantId);
  }

  /**
   * Delete a role
   */
  async deleteRole(roleId, tenantId) {
    // Check if it's a system role
    const role = await this.getRole(roleId, tenantId);
    if (!role) return false;
    
    if (role.is_system_role) {
      throw new Error('Cannot delete system role');
    }
    
    await db.query(
      'DELETE FROM roles WHERE id = ? AND tenant_id = ?',
      [roleId, tenantId]
    );
    
    return true;
  }

  /**
   * Set permissions for a role
   */
  async setRolePermissions(roleId, permissionIds) {
    const connection = await db.getConnection();
    
    try {
      await connection.beginTransaction();
      
      // Delete existing permissions
      await connection.query(
        'DELETE FROM role_permissions WHERE role_id = ?',
        [roleId]
      );
      
      // Add new permissions
      for (const permissionId of permissionIds) {
        await connection.query(
          'INSERT INTO role_permissions (id, role_id, permission_id) VALUES (?, ?, ?)',
          [uuidv4(), roleId, permissionId]
        );
      }
      
      await connection.commit();
      return true;
    } catch (error) {
      await connection.rollback();
      throw error;
    } finally {
      connection.release();
    }
  }
}

module.exports = new RoleService();
```

### User Role Service

```javascript
// services/userRoleService.js

const { v4: uuidv4 } = require('uuid');
const db = require('../config/database');

class UserRoleService {
  /**
   * Get roles assigned to a user
   */
  async getUserRoles(userId) {
    const query = `
      SELECT ur.id, ur.role_id, r.name as role_name, 
             ur.scope, ur.scope_id,
             CASE 
               WHEN ur.scope = 'store' THEN s.name 
               WHEN ur.scope = 'tenant' THEN t.name 
               ELSE NULL 
             END as scope_name
      FROM user_roles ur
      JOIN roles r ON ur.role_id = r.id
      LEFT JOIN stores s ON ur.scope = 'store' AND ur.scope_id = s.id
      LEFT JOIN tenants t ON ur.scope = 'tenant' AND ur.scope_id = t.id
      WHERE ur.user_id = ?
    `;
    
    const [rows] = await db.query(query, [userId]);
    return rows;
  }

  /**
   * Assign a role to a user
   */
  async assignUserRole(userId, roleId, scope, scopeId) {
    // Validate the role exists
    const [roleRows] = await db.query(
      'SELECT * FROM roles WHERE id = ?',
      [roleId]
    );
    
    if (!roleRows.length) {
      throw new Error('Role not found');
    }
    
    // Create the assignment
    const id = uuidv4();
    
    await db.query(
      'INSERT INTO user_roles (id, user_id, role_id, scope, scope_id) VALUES (?, ?, ?, ?, ?)',
      [id, userId, roleId, scope, scopeId]
    );
    
    return { id, userId, roleId, scope, scopeId };
  }

  /**
   * Remove a role from a user
   */
  async removeUserRole(userRoleId, userId) {
    const [result] = await db.query(
      'DELETE FROM user_roles WHERE id = ? AND user_id = ?',
      [userRoleId, userId]
    );
    
    return result.affectedRows > 0;
  }

  /**
   * Get stores accessible by a user
   */
  async getUserAccessibleStores(userId, tenantId) {
    // Get all stores the user has direct access to
    const query = `
      SELECT DISTINCT s.id, s.name, r.name as role_name
      FROM stores s
      JOIN user_roles ur ON ur.scope = 'store' AND ur.scope_id = s.id
      JOIN roles r ON ur.role_id = r.id
      WHERE s.tenant_id = ? AND ur.user_id = ?

      UNION

      -- Include all stores if user has tenant-level role
      SELECT s.id, s.name, r.name as role_name
      FROM stores s
      CROSS JOIN user_roles ur
      JOIN roles r ON ur.role_id = r.id
      WHERE s.tenant_id = ? 
        AND ur.user_id = ? 
        AND ur.scope = 'tenant'
    `;
    
    const [rows] = await db.query(query, [tenantId, userId, tenantId, userId]);
    return rows;
  }
}

module.exports = new UserRoleService();
```

## Frontend Implementation Examples

### Permission Hook

```typescript
// src/hooks/usePermission.ts
import { useAuth } from '../context/AuthContext';

export const usePermission = () => {
  const { permissions, currentStoreId } = useAuth();

  /**
   * Check if the current user has a specific permission
   * @param permission Permission name or array of permissions
   * @param requireAll If true, user must have all permissions when an array is provided
   */
  const hasPermission = (
    permission: string | string[],
    requireAll: boolean = false
  ): boolean => {
    // No permissions loaded yet
    if (!permissions || !permissions.length) return false;

    // Handle array of permissions
    if (Array.isArray(permission)) {
      if (requireAll) {
        return permission.every(p => permissions.includes(p));
      } else {
        return permission.some(p => permissions.includes(p));
      }
    }

    // Single permission check
    return permissions.includes(permission);
  };

  return {
    hasPermission,
    currentStoreId
  };
};
```

```tsx
// src/components/Navigation.tsx
import React from 'react';
import { usePermission } from '../hooks/usePermission';
import { Link } from 'react-router-dom';

interface NavItem {
  label: string;

  icon?: React.ReactNode;
  children?: NavItem[];
}

const navItems: NavItem[] = [
  {
    label: 'Dashboard',
    path: '/dashboard',
    icon: <DashboardIcon />
  },
  {
    label: 'Products',
    path: '/products',
    permission: 'products.view',
    icon: <ProductsIcon />

    label: 'Inventory',
    path: '/inventory',
    permission: 'inventory.view',
    icon: <InventoryIcon />
  },
  {
    label: 'Sales',
    path: '/sales',
    permission: 'sales.view',
    icon: <SalesIcon />
  },

    label: 'Customers',
    path: '/customers',
    permission: 'customers.view',
    icon: <CustomersIcon />
  },
  {
    label: 'Reports',
    path: '/reports',
    permission: ['reports.sales', 'reports.inventory', 'reports.financial'],
    icon: <ReportsIcon />,
    children: [
      {
        label: 'Sales Reports',
        path: '/reports/sales',
        permission: 'reports.sales'
      },
      {
        label: 'Inventory Reports',
        path: '/reports/inventory',
        permission: 'reports.inventory'
      },
      {
        label: 'Financial Reports',
        path: '/reports/financial',
        permission: 'reports.financial'
      }
    ]
  },
  {
    label: 'Settings',
    path: '/settings',
    permission: 'settings.view',
    icon: <SettingsIcon />
  },
  {
    label: 'User Management',
    path: '/users',
    permission: 'users.view',
    icon: <UsersIcon />
  }
];

export const Navigation: React.FC = () => {
  const { hasPermission } = usePermission();

  const renderNavItems = (items: NavItem[]) => {
    return items.map((item) => {
      // Skip items that require permissions the user doesn't have
      if (item.permission && !hasPermission(item.permission)) {
        return null;
      }

      return (
        <li key={item.path}>
          <Link to={item.path} className="nav-link">
            {item.icon && <span className="icon">{item.icon}</span>}
            <span className="label">{item.label}</span>
          </Link>
          
          {item.children && item.children.length > 0 && (
            <ul className="submenu">
              {renderNavItems(item.children)}
            </ul>
          )}
        </li>
      );
    });
  };

  return (
    <nav className="main-navigation">
      <ul>{renderNavItems(navItems)}</ul>
    </nav>
  );
};
```

### PermissionGuard HOC

```tsx
// src/components/PermissionGuard.tsx
import React from 'react';
import { Navigate } from 'react-router-dom';
import { usePermission } from '../hooks/usePermission';

interface PermissionGuardProps {
  permission: string | string[];
  requireAll?: boolean;
  redirectTo?: string;
  children: React.ReactNode;
}

export const PermissionGuard: React.FC<PermissionGuardProps> = ({
  permission,
  requireAll = false,
  redirectTo = '/unauthorized',
  children
}) => {
  const { hasPermission } = usePermission();
  
  if (hasPermission(permission, requireAll)) {
    return <>{children}</>;
  }
  
  return <Navigate to={redirectTo} />;
};

// Usage in routes
const AppRoutes = () => (
  <Routes>
    <Route path="/dashboard" element={<Dashboard />} />
    
    <Route
      path="/products"
      element={
        <PermissionGuard permission="products.view">
          <ProductsPage />
        </PermissionGuard>
      }
    />
    
    <Route
      path="/settings"
      element={
        <PermissionGuard permission="settings.view">
          <SettingsPage />
        </PermissionGuard>
      }
    />
    
    <Route path="/unauthorized" element={<UnauthorizedPage />} />
  </Routes>
);
```

## Role and Permission Management UI Example

```tsx
// src/pages/RoleManagement.tsx
import React, { useState, useEffect } from 'react';
import { fetchApi } from '../utils/api';
import { usePermission } from '../hooks/usePermission';
import { Permission } from '../components/Permission';

interface Role {
  id: string;
  name: string;
  description: string;
  is_system_role: boolean;
}

interface PermissionItem {
  id: string;
  name: string;
  description: string;
  module: string;
}

export const RoleManagement: React.FC = () => {
  const [roles, setRoles] = useState<Role[]>([]);
  const [selectedRole, setSelectedRole] = useState<Role | null>(null);
  const [permissions, setPermissions] = useState<PermissionItem[]>([]);
  const [rolePermissions, setRolePermissions] = useState<string[]>([]);
  const [permissionsByModule, setPermissionsByModule] = useState<Record<string, PermissionItem[]>>({});
  
  const { hasPermission } = usePermission();
  
  // Fetch roles on load
  useEffect(() => {
    const fetchRoles = async () => {
      try {
        const response = await fetchApi('/api/roles');
        setRoles(response.data || []);
      } catch (error) {
        console.error('Failed to fetch roles:', error);
      }
    };
    
    fetchRoles();
  }, []);
  
  // Fetch all permissions
  useEffect(() => {
    const fetchPermissions = async () => {
      try {
        const response = await fetchApi('/api/permissions');
        setPermissions(response.data || []);
        
        // Group by module
        const grouped = (response.data || []).reduce((acc: any, curr: PermissionItem) => {
          if (!acc[curr.module]) {
            acc[curr.module] = [];
          }
          acc[curr.module].push(curr);
          return acc;
        }, {});
        
        setPermissionsByModule(grouped);
      } catch (error) {
        console.error('Failed to fetch permissions:', error);
      }
    };
    
    fetchPermissions();
  }, []);
  
  // Fetch role permissions when a role is selected
  useEffect(() => {
    const fetchRolePermissions = async () => {
      if (!selectedRole) return;
      
      try {
        const response = await fetchApi(`/api/roles/${selectedRole.id}/permissions`);
        setRolePermissions(response.data.map((p: PermissionItem) => p.id) || []);
      } catch (error) {
        console.error('Failed to fetch role permissions:', error);
      }
    };
    
    fetchRolePermissions();
  }, [selectedRole]);
  
  const handleRoleSelect = (role: Role) => {
    setSelectedRole(role);
  };
  
  const handlePermissionToggle = (permissionId: string) => {
    if (rolePermissions.includes(permissionId)) {
      setRolePermissions(rolePermissions.filter(id => id !== permissionId));
    } else {
      setRolePermissions([...rolePermissions, permissionId]);
    }
  };
  
  const savePermissions = async () => {
    if (!selectedRole) return;
    
    try {
      await fetchApi(`/api/roles/${selectedRole.id}/permissions`, {
        method: 'PUT',
        body: JSON.stringify({ permissionIds: rolePermissions })
      });
      
      alert('Permissions saved successfully');
    } catch (error) {
      console.error('Failed to save permissions:', error);
      alert('Failed to save permissions');
    }
  };
  
  return (
    <div className="role-management">
      <h1>Role Management</h1>
      
      <div className="role-management-container">
        <div className="roles-list">
          <h2>Roles</h2>
          <ul>
            {roles.map(role => (
              <li 
                key={role.id} 
                className={selectedRole?.id === role.id ? 'selected' : ''}
                onClick={() => handleRoleSelect(role)}
              >
                {role.name}
                {role.is_system_role && <span className="badge">System</span>}
              </li>
            ))}
          </ul>
          
          <Permission require="roles.manage">
            <button>Add New Role</button>
          </Permission>
        </div>
        
        {selectedRole && (
          <div className="role-details">
            <h2>{selectedRole.name}</h2>
            <p>{selectedRole.description}</p>
            
            <div className="permissions-section">
              <h3>Permissions</h3>
              
              {Object.entries(permissionsByModule).map(([module, modulePermissions]) => (
                <div key={module} className="permission-module">
                  <h4>{module}</h4>
                  <div className="permission-list">
                    {modulePermissions.map(permission => (
                      <div key={permission.id} className="permission-item">
                        <label>
                          <input 
                            type="checkbox"
                            checked={rolePermissions.includes(permission.id)}
                            onChange={() => handlePermissionToggle(permission.id)}
                            disabled={!hasPermission('roles.manage') || selectedRole.is_system_role}
                          />
                          {permission.name}
                        </label>
                        <span className="permission-description">{permission.description}</span>
                      </div>
                    ))}
                  </div>
                </div>
              ))}
              
              <Permission require="roles.manage">
                <button 
                  onClick={savePermissions}
                  disabled={selectedRole.is_system_role}
                >
                  Save Permissions
                </button>
              </Permission>
            </div>
          </div>
        )}
      </div>
    </div>
  );
};
```
